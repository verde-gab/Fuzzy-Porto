	\begin{frame}{O Caso-Problema: \textcite{santos2023porto}}
		
		\textcite{santos2023porto} analisam a movimentação de carga do Porto de Santos
		(2005--2022) e propõem um modelo de previsão para 2023--2026 baseado em
		\textbf{regressão linear simples}.
		
	\begin{figure}
		\centering
		\includegraphics[width=1\linewidth]{Images/real_rl.png}
	\end{figure}
		
	\end{frame}
	
	\begin{frame}{A Simplicidade do Modelo como Motivação}
		
		Os autores justificam a escolha afirmando que, embora existam modelos
		mais adequados para séries temporais com sazonalidade (e.g., SARIMA),
		a regressão linear oferece:
		
		\vspace{5mm}
		
		\begin{quote}
			\textit{``[...] um modelo simples como a regressão linear em uma
				predição e realizando-o com um custo computacional muito menor.''}
			\hfill \parencite{santos2023porto}
		\end{quote}
		
	\end{frame}
	
	
	\begin{frame}{A Estratégia dos 12 Modelos Mensais}
		
		Para contornar a sazonalidade, os autores treinaram
		\textbf{12 regressões lineares independentes} — uma para cada mês:
		
		\begin{figure}
			\centering
			\includegraphics[width=0.86\linewidth]{Images/tabela.png} \\
			\footnotesize {\tiny Fonte: \textcite{santos2023porto}}
		\end{figure}
		
	\end{frame}
	
		\begin{frame}
		
		\begin{quote}
			\textit{"Diante das correlações encontradas para cada uma das curvas individualizadas dos meses serem aceitáveis, bem como os coeficientes de determinação ($R^{2} \ge 80\%$), seguiu-se com a determinação da reta média da regressão linear."}
			\hfill \parencite{santos2023porto}
		\end{quote}
		
		\vspace{5mm}
		
		Os altos $R^2$ são apresentados como evidência de ``sucesso''
		do modelo — mas será isso suficiente?
		
	\end{frame}
	
	\begin{frame}{Regressão Espúria e Ajuste de Curva}                                                                                                                 
		\textcite{granger1974spurious} demonstram justamente esse padrão onde            nos modelos de Regressão Linear podem-se ter coeficientes $R^{2}$ altos e ao mesmo tempo não passam no teste Durbin-Watson para autocorrelação, sendo esses modelos classificados como \textbf{Regressões Espúrias}.                                                               
		
		\vspace{1em}                                                                                                                                 
		Na concepção de \textcite{spanos2007curve}, o caso \textcite{santos2023porto} representa um problema de \textit{Curve Fitting} (Ajuste de Curva). Ao tratarem a solução através da métrica do coeficiente $R^{2}$, a avaliação de sucesso do modelo foi definida como \textit{goodness-of-fit} (Qualidade do Ajuste), em uma aplicação de 12 modelos com $R^{2} \geq 80\%$.
		
	\end{frame}
	
	\begin{frame}{Questões Primárias e Secundárias}
		\textcite{mayo2004methodology} distinguem dois níveis
		de perguntas em um modelo estatístico:
		
		\vspace{0.2mm}
		
		\begin{block}{Questão Primária (dentro do modelo)}
			Dado que o modelo é válido, quão bom preditor é $x$? \quad $\Rightarrow$ $R^2$, p-valores, coeficientes... Conforme vimos em \textcite{santos2023porto}.
		\end{block}
		
		\begin{block}{Questão Secundária (sobre o modelo)}
			Os dados $z$ satisfazem as premissas probabilísticas
			do modelo? \quad $\Rightarrow$ testes de misspecification
		\end{block}
		
		\vspace{0.2mm}
		
		\begin{quote}
			\footnotesize\textit{``We can answer [if inference is reliable]
				affirmatively only if data $z$ satisfy the probabilistic
				assumptions of the LRM, i.e., the errors are NIID.''}
			\hfill \parencite{mayo2004methodology}
		\end{quote}
		
	\end{frame}
	
	\begin{frame}{Adequação Estatística}
		\textcite{spanos2007curve} define Adequação Estatística como o
		critério para avaliar se as premissas assumidas por um modelo
		probabilístico realmente valem para os dados observados — no
		exemplo de uma regressão linear, a verificação de
		Homocedasticidade, Independência, Normalidade... Portanto traz
		um critério procedural e normativo no sentido dos modelos
		clássicos estocásticos.
		
		\vspace{5mm}
		
		\begin{quote}
			\textit{``Statistical adequacy is tantamount to the claim
				that data $y_0$ constitute a `truly typical realization' of
				the stochastic mechanism described by $M_v(y)$.''}
			\hfill \parencite{spanos2007curve}
		\end{quote}
		
	\end{frame}

			\end{column}
		\end{columns}
	\end{frame}

	\begin{frame}
		A regressão linear pressupõe resíduos \textbf{independentes e
		identicamente distribuídos} ($\varepsilon_i \overset{\text{iid}}{\sim}
		\mathcal{N}(0, \sigma^2)$).

		\end{block}
		
		No critério de \textcite{mayo2018statistical} isso representa um \textbf{BENT} (\textit{Bad Evidence, No Test}).

	\begin{frame}{Newton C. A. da Costa e o "Pluralismo Lógico"}
		
		Newton da Costa já denotava a existência de diversas lógicas, e a lógica sendo uma ferramenta do ser humano para mapear a realidade. Diferentes lógicas servem a diferentes problemas, sem que uma seja intrinsecamente "mais verdadeira" que outra.
		
		\vspace{5mm}
		
		Aqui iremos nos apoiar nesses autores e trocar o paradigma dos modelos clássicos/estocásticos para demonstrar que os modelos que escolhemos vem acompanhados de um compromisso formal e compostos de uma estrutura subjacente.
		
	\end{frame}

	\begin{frame}{Teoria de Conjuntos Fuzzy}
		Depois de \textcite{zadeh1965fuzzy} consolidar a teoria de conjuntos Fuzzy, muito se desenvolveu no âmbito de problemas com variáveis linguísticas/subjetivas. A lógica Fuzzy estende a lógica booleana para os conjuntos com graus de pertinência onde uma mesma variável pode ser parte verdadeira e parte falsa ao mesmo tempo.

	\begin{frame}{A Metamatemática da Lógica Fuzzy por \textcite{hajek1998metamathematics}}
		\begin{columns}[T,onlytextwidth]
			\begin{column}{0.35\textwidth}
				\includegraphics[width=0.9\linewidth]{Images/hajek}
				
			\footnotesize
			\textcite{hajek1998metamathematics} constitui em seu livro a \textbf{base formal e metamatemática da lógica fuzzy}, elevando-a ao status de um sistema lógico rigoroso.
			\end{column}
			
			\begin{column}{0.60\textwidth}
				Nesta obra seminal, o autor demonstra que:
				\begin{itemize}
					\item O raciocínio aproximado (vago) possui fundamentos matemáticos sólidos, com sintaxe, semântica e completude bem definidas.
					\item O sistema formal consolida-se sobre lógicas polivalentes infinitas (baseadas em \textit{t-normas}).
					\item A teoria tem como um de seus pilares centrais a pioneira \textbf{Lógica de Jan Łukasiewicz}.
				\end{itemize}
			\end{column}
			
		\end{columns}
		
	\end{frame}
	
	\begin{frame}{Fuzzy Time Series}
		Introduzido por \textcite{song1993fuzzy} os modelos \textit{Fuzzy Time Series}, baseados na lógica Fuzzy de \textcite{zadeh1965fuzzy}, apresentam uma ferramenta formal para a previsão de séries temporais.
	
		\vspace{5mm}
		
		O modelo inicial de \autocite{song1993fuzzy} apresentava um custo computacional muito grande devido as operações matriciais e posteriormente foi simplificado por \textcite{chen1996forecasting} que substituiu por operações aritméticas simples.
		
		\begin{table}[]
			\centering
			\renewcommand{\arraystretch}{1.2}
			\resizebox{\columnwidth}{!}{%
				\begin{tabular}{l|c|c}
					& \textbf{Song \& Chissom (1993)} & \textbf{Chen (1996)} \\ \hline
					\textbf{Relações} 
					& $R = \bigcup_{i=1}^{n} A_{i}^{\top} \times A_{i+1}$ 
					& Grupos de Relações Lógicas Fuzzy \\ \hline
					\textbf{Previsões} 
					& $A_i = A_{i-1} \circ R$ 
					& $\hat{y}_{i+1} = \frac{1}{p}\sum_{k=1}^{p} m_k$
				\end{tabular}%
			}
		\end{table}
	
	\end{frame}

	\begin{frame}{Honestidade Axiomática}
		
		\begin{block}{Definição}                                                                                                                     
			Um modelo é \textbf{axiomaticamente honesto} quando                                                                                      
			nenhuma afirmação inferencial que dele se extrai excede                                                                                  
			o que suas premissas formais autorizam --- seja porque                                                                                   
			essas premissas foram testadas e não rejeitadas, seja                                                                                    
			porque, na ausência de teste, a afirmação foi                                                                                            
			correspondentemente contida.                                                                                                             
		\end{block}
		
		\vspace{0.4em}
		
		\centering
		\footnotesize
		\begin{tabular}{l c c}
			\hline
			\textbf{Modelo} & \textbf{Premissas} & \textbf{Honesto?} \\
			\hline
			RL \parencite{santos2023porto}
			& iid rejeitada ($\text{LB}_{12}\!=\!119{,}27$)
			& \textbf{Não} \\
			Chen FTS (sobre $\Delta y$)
			& sem premissa de iid
			& \textbf{Sim} \\
			\hline
		\end{tabular}
		
		\vspace{0.5em}
		
		\begin{quote}
			\footnotesize\textit{O critério é universal: qualquer
				modelo que respeite suas próprias premissas é
				axiomaticamente honesto --- independentemente do
				paradigma formal em que opera.}
		\end{quote}
		
	\end{frame}
